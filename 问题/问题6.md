// 引入 Rust 的标准库
use std::time::{SystemTime, UNIX_EPOCH};
// 引入 nalgebra 库的相关组件
use nalgebra::{Matrix3, MatrixXd, U3};
fn main() {
    // 获取一个高质量的随机种子
    let rd = SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_secs();
    let mut rng = rand::thread_rng(); // 初始化随机数生成器
    let distribution = rand::distributions::Uniform::new(-1.0, 1.0); // 均匀分布 [-1, 1]
    // 设置矩阵的大小
    let matrix_size = 5;
    // 初始化 nalgebra 矩阵
    let mut big_matrix = MatrixXd::new_random(matrix_size, matrix_size, &mut rng, &distribution);
    // 使用随机数填充矩阵
    for i in 0..matrix_size {
        for j in 0..matrix_size {
            big_matrix[(i, j)] = distribution.sample(&mut rng); // 填充每个元素
        }
    }
    println!("The big matrix:\n{:?}", big_matrix);
    // 提取 3x3 的矩阵块
    let extracted_block = big_matrix.fixed_slice::<U3, U3>(0, 0);
    println!("The extracted matrix block:\n{:?}", extracted_block);
    // 将提取的矩阵块设置为单位矩阵
    let mut assigned_block = Matrix3::identity();
    for i in 0..3 {
        for j in 0..3 {
            assigned_block[(i, j)] = extracted_block[(i, j)];
        }
    }
    println!("The assigned matrix block:\n{:?}", assigned_block);
}

1. **随机数生成**：Rust中使用`rand`库生成随机数，这里使用了`thread_rng`作为随机数生成器，并定义了均匀分布。
2. **矩阵操作**：使用`nalgebra`库的`MatrixXd`和`Matrix3`来表示动态大小矩阵和3x3矩阵。`fixed_slice`方法用于提取矩阵的子块。
3. **打印**：Rust中使用`println!`宏来打印信息，格式化为`{:?}`以友好地显示矩阵内容。
4. **错误处理**：`unwrap`用于处理`SystemTime::now().duration_since(UNIX_EPOCH)`可能返回的`None`情况，这里假设系统时间不会早于UNIX纪元。
