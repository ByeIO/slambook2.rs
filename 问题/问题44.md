    }

#![allow(dead_code)]
#![allow(non_snake_case)]

use super::camera::Camera;
use super::preclude::*;
use std::sync::{Arc, Mutex, Weak};
use std::cell::RefCell;
use cv::Mat;

// 前置声明
pub struct MapPoint;
pub struct Feature;

/// 帧
/// 每一帧分配独立id，关键帧分配关键帧ID
pub struct Frame {
    /// 此帧的唯一 ID
    pub id_: u64,
    /// 关键帧的 ID
    pub keyframe_id_: u64,
    /// 是否为关键帧
    pub is_keyframe_: bool,

}

impl Frame {
    /// 默认构造函数
    pub fn new() -> Self {
        Frame {
            id_: 0,
            keyframe_id_: 0,
            is_keyframe_: false,
            time_stamp_: 0.0,
            pose_: SE3::identity(),
            pose_mutex_: Mutex::new(()),
            left_img_: Mat::default(),
            right_img_: Mat::default(),
            features_left_: RefCell::new(Vec::new()),
            features_right_: RefCell::new(Vec::new()),
        }
    }

    pub fn with_params(id: u64, time_stamp: f64, pose: SE3, left: Mat, right: Mat) -> Self {
        Frame {
            id_,
            keyframe_id_: 0,
            is_keyframe_: false,
            time_stamp_,
            pose,
            pose_mutex_: Mutex::new(()),
            left_img_: left,
            right_img_: right,
            features_left_: RefCell::new(Vec::new()),
            features_right_: RefCell::new(Vec::new()),
        }
    }

    /// 获取帧的位姿，线程安全
    pub fn Pose(&self) -> SE3 {
        let _lck = self.pose_mutex_.lock().unwrap();
        self.pose_.clone()
    }

    /// 设置帧的位姿，线程安全
    pub fn SetPose(&self, pose: &SE3) {
        let _lck = self.pose_mutex_.lock().unwrap();
        self.pose_ = pose.clone();
    }

    /// 设置关键帧并分配关键帧 ID
    pub fn SetKeyFrame(&mut self) {
        static mut KEYFRAME_FACTORY_ID: u64 = 0;
        self.is_keyframe_ = true;
        unsafe {
            self.keyframe_id_ = KEYFRAME_FACTORY_ID;
            KEYFRAME_FACTORY_ID += 1;
        }
    }

    /// 工厂构建模式，分配 ID
    pub fn CreateFrame() -> Arc<Frame> {
        static mut FACTORY_ID: u64 = 0;
        let new_frame = Arc::new(Frame::new());
        unsafe {
            Arc::get_mut(&mut Arc::clone(&new_frame)).unwrap().id_ = FACTORY_ID;
            FACTORY_ID += 1;
        }
        new_frame
    }
}

### 代码说明：
1. **结构体定义**：`Frame` 结构体包含了帧的基本信息，如 `id_`、`keyframe_id_`、`is_keyframe_` 等，以及左右图像和特征向量。
2. **构造函数**：提供了默认构造函数 `new` 和带参数的构造函数 `with_params`。
3. **线程安全的位姿访问**：`Pose` 方法用于获取帧的位姿，`SetPose` 方法用于设置帧的位姿，通过 `Mutex` 保证线程安全。
4. **设置关键帧**：`SetKeyFrame` 方法将帧设置为关键帧，并分配唯一的关键帧 ID。
5. **工厂模式创建帧**：`CreateFrame` 方法使用工厂模式创建新的帧，并分配唯一的 ID。

需要注意的是，Rust 中没有像 C++ 那样的 `std::shared_ptr`，我们使用 `Arc` 来实现类似的功能，同时使用 `Mutex` 来保证线程安全。另外，由于 Rust 的所有权和借用规则，我们使用 `RefCell` 来实现内部可变性。

        let r_mat = self.rot.to_matrix();
        let t_r_mat = SO3::hat(self.xyz.as_view()) * r_mat;

        mat.fixed_view_mut::<3, 3>(0, 0).copy_from(&r_mat);
        mat.fixed_view_mut::<3, 3>(3, 3).copy_from(&r_mat);
        mat.fixed_view_mut::<3, 3>(3, 0).copy_from(&t_r_mat);

        mat
    }

    fn hat(xi: VectorView6<T>) -> Matrix4<T> {
        let mut mat = Matrix4::zeros();
        mat[(0, 1)] = -xi[2];
        mat[(0, 2)] = xi[1];
        mat[(1, 0)] = xi[2];
        mat[(1, 2)] = -xi[0];
        mat[(2, 0)] = -xi[1];
        mat[(2, 1)] = xi[0];

        mat[(0, 3)] = xi[3];
        mat[(1, 3)] = xi[4];
        mat[(2, 3)] = xi[5];

        mat
    }

    fn vee(xi: MatrixView<4, 4, T>) -> Vector6<T> {
        Vector6::new(
            xi[(2, 1)],
            xi[(0, 2)],
            xi[(1, 0)],
            xi[(0, 3)],
            xi[(1, 3)],
            xi[(2, 3)],
        )
    }

    fn hat_swap(xi: VectorView3<T>) -> Matrix3x6<T> {
        let mut mat = Matrix3x6::zeros();
        mat.fixed_view_mut::<3, 3>(0, 0)
            .copy_from(&SO3::hat_swap(xi.as_view()));
        mat.fixed_view_mut::<3, 3>(0, 3)
            .copy_from(&Matrix3::identity());
        mat
    }

    fn apply(&self, v: VectorView3<T>) -> Vector3<T> {
        self.rot.apply(v) + self.xyz
    }

    fn to_matrix(&self) -> Matrix4<T> {
        let mut mat = Matrix4::<T>::identity();
        mat.fixed_view_mut::<3, 3>(0, 0)
            .copy_from(&self.rot.to_matrix());
        mat.fixed_view_mut::<3, 1>(0, 3).copy_from(&self.xyz);
        mat
    }

    fn from_matrix(mat: MatrixView<4, 4, T>) -> Self {
        let rot = mat.fixed_view::<3, 3>(0, 0).clone_owned();
        let rot = SO3::from_matrix(rot.as_view());

        let xyz = mat.fixed_view::<3, 1>(0, 3).into();

        SE3 { rot, xyz }
    }
}

    use super::*;
    use crate::linalg::{Vector3, Matrix3};

    // 辅助函数，用于比较两个 Vector3 是否近似相等
    fn vector3_approx_eq<T: Numeric>(a: &Vector3<T>, b: &Vector3<T>, abs: T, rel: T) -> bool {
        let diff = (a - b).norm();
        let tolerance = rel * a.norm().max(b.norm()) + abs;
        diff <= tolerance
    }

    // 辅助函数，用于比较两个 Matrix3 是否近似相等
    fn matrix3_approx_eq<T: Numeric>(a: &Matrix3<T>, b: &Matrix3<T>, abs: T, rel: T) -> bool {
        let mut diff_sum = T::zero();
        for i in 0..3 {
            for j in 0..3 {
                let diff = (a[(i, j)] - b[(i, j)]).abs();
                let tolerance = rel * a[(i, j)].abs().max(b[(i, j)].abs()) + abs;
                if diff > tolerance {
                    return false;
                }
                diff_sum += diff;
            }
        }
        diff_sum <= rel * a.frobenius_norm().max(b.frobenius_norm()) + abs
    }

    fn vector3_approx_eq<T: Numeric>(a: &Vector3<T>, b: &Vector3<T>, abs: T, rel: T) -> bool {
        let diff = (a - b).norm();
        let tolerance = rel * a.norm().max(b.norm()) + abs;
        diff <= tolerance
    }

    // 辅助函数，用于比较两个 Matrix3 是否近似相等
    fn matrix3_approx_eq<T: Numeric>(a: &Matrix3<T>, b: &Matrix3<T>, abs: T, rel: T) -> bool {
        let mut diff_sum = T::zero();
        for i in 0..3 {
            for j in 0..3 {
                let diff = (a[(i, j)] - b[(i, j)]).abs();
                let tolerance = rel * a[(i, j)].abs().max(b[(i, j)].abs()) + abs;
                if diff > tolerance {
                    return false;
                }
                diff_sum += diff;
            }
        }

    }

/// 数据集读取
/// 构造时传入配置文件路径，配置文件的dataset_dir为数据集路径

pub struct Dataset {

    current_image_index: usize,
    // 相机列表
    cameras: Vec<Arc<Camera>>,

impl Dataset {
    // 定义类型别名，方便后续使用
    pub type Ptr = Arc<Self>;

    /// 构造函数，接收数据集路径作为参数
    pub fn new(dataset_path: &str) -> Self {
        Dataset {
            dataset_path: dataset_path.to_string(),
            current_image_index: 0,
            cameras: Vec::new(),
        }
    }

    /// 初始化，返回是否成功
    pub fn init(&mut self) -> bool {
        // 打开相机校准文件
        let calib_path = Path::new(&self.dataset_path).join("calib.txt");
        let file = match File::open(calib_path) {
            Ok(file) => file,
            Err(_) => {
                error!("cannot find {}/calib.txt!", self.dataset_path);
                return false;
            }
        };
        let reader = BufReader::new(file);
        let mut lines = reader.lines();

        for i in 0..4 {
            // 读取相机名称
            let camera_name = lines.next().unwrap().unwrap();
            // 读取投影数据
            let projection_line = lines.next().unwrap().unwrap();
            let mut projection_data: Vec<f64> = projection_line.split_whitespace()
                .map(|s| s.parse().unwrap())
                .collect();

            // 构建相机内参矩阵 K
            let mut K = Matrix3::zeros();
            K[(0, 0)] = projection_data[0];
            K[(0, 1)] = projection_data[1];
            K[(0, 2)] = projection_data[2];
            K[(1, 0)] = projection_data[4];
            K[(1, 1)] = projection_data[5];
            K[(1, 2)] = projection_data[6];
            K[(2, 0)] = projection_data[8];
            K[(2, 1)] = projection_data[9];
            K[(2, 2)] = projection_data[10];

            // 构建平移向量 t
            let t = Vector3::new(projection_data[3], projection_data[7], projection_data[11]);
            let t = K.try_inverse().unwrap() * t;
            K = K * 0.5;

            // 创建新的相机实例
            let new_camera = Arc::new(Camera::with_params(
                K[(0, 0)], K[(1, 1)], K[(0, 2)], K[(1, 2)],
                t.norm(), SE3::new(SO3::identity(), t),
            ));
            self.cameras.push(new_camera);
            info!("Camera {} extrinsics: {:?}", i, t.transpose());
        }

        self.current_image_index = 0;
        true
    }

    /// 创建并返回包含立体图像的下一帧
    pub fn next_frame(&mut self) -> Option<Arc<Frame>> {
        // 生成图像文件路径
        let mut left_path = String::new();
        let mut right_path = String::new();
        write!(&mut left_path, "{}/image_0/{:06}.png", self.dataset_path, self.current_image_index).unwrap();
        write!(&mut right_path, "{}/image_1/{:06}.png", self.dataset_path, self.current_image_index).unwrap();

        // 读取左右图像
        let image_left = match cv::imread(&left_path, cv::IMREAD_GRAYSCALE) {
            Ok(img) => img,
            Err(_) => {
                warn!("cannot find images at index {}", self.current_image_index);
                return None;
            }
        };
        let image_right = match cv::imread(&right_path, cv::IMREAD_GRAYSCALE) {
            Ok(img) => img,
            Err(_) => {
                warn!("cannot find images at index {}", self.current_image_index);
                return None;
            }
        };

        // 调整图像大小
        let mut image_left_resized = OMatrix::default();
        let mut image_right_resized = OMatrix::default();
        cv::resize(&image_left, &mut image_left_resized, cv::Size::default(), 0.5, 0.5, cv::INTER_NEAREST);
        cv::resize(&image_right, &mut image_right_resized, cv::Size::default(), 0.5, 0.5, cv::INTER_NEAREST);

        // 创建新的帧
        let new_frame = Frame::CreateFrame();
        new_frame.left_img_ = image_left_resized;
        new_frame.right_img_ = image_right_resized;
        self.current_image_index += 1;

        Some(new_frame)
    }
