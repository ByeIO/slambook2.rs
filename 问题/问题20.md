```rs
// 读取相机位姿
fn read_poses(path: &str) -> TrajectoryType {
    let file = File::open(path).expect("pose.txt not found.");
    let reader = BufReader::new(file);
    let mut poses = Vec::new();

    for line in reader.lines() {
        let line = line.expect("Failed to read line");
        let data: Vec<f64> = line.split_whitespace().map(|s| s.parse().unwrap()).collect();
        if data.len() == 7 {
            // 创建单位四元数
            let quaternion = UnitQuaternion::from_quaternion(Quaternion::new(data[6], data[3], data[4], data[5]));
            // 创建平移
            let translation = Translation3::new(data[0], data[1], data[2]);
            // 创建李代数
            let isometry = Isometry3::<f64>::from_parts(translation, quaternion);
            poses.push(isometry);
        }
    }
    // 返回位姿
    poses
}

// 加载图像
fn load_images(num: usize) -> (Vec<DynamicImage>, Vec<DynamicImage>) {

}

// 计算点云
fn compute_pointcloud(color_imgs: &Vec<DynamicImage>, depth_imgs: &Vec<DynamicImage>, poses: &TrajectoryType) -> Vec<Vector6d> {
    let cx = 325.5;
    let cy = 253.5;
    let fx = 518.0;
    let fy = 519.0;
    let depth_scale = 1000.0;
    let mut pointcloud = Vec::new();

    for (i, color) in color_imgs.iter().enumerate() {
        let depth = &depth_imgs[i];
        let T = poses[i];

        for v in 0..color.height() {
            for u in 0..color.width() {
                // 获取深度值
                let d = depth.get_pixel(u, v)[0]; 
                // 深度值为0表示没有测量到
                if d == 0 { continue; } 

                let mut point = Vector3::zeros();
                point[2] = d as f64 / depth_scale;
                point[0] = (u as f64 - cx) * point[2] / fx;
                point[1] = (v as f64 - cy) * point[2] / fy;

                let point_world = T * point;

                let mut p = Vector6d::zeros();

                // 从point_world中复制数据到6x1矩阵
                p.fixed_rows_mut::<3>(0).copy_from(&point_world.into());

                let color_data = color.get_pixel(u, v);
                // red
                p[3] = color_data[2] as f64; 
                // green
                p[4] = color_data[1] as f64; 
                // blue
                p[5] = color_data[0] as f64; 
                pointcloud.push(p);
            }
        }
    }
    pointcloud
}
```

以下是StereoSGBM（Semi-Global Block Matching）算法的伪代码，用于计算立体视觉中的视差图：

### StereoSGBM 算法伪代码

```plaintext
输入:
    left_image: 左图像
    right_image: 右图像
    min_disparity: 最小视差
    max_disparity: 最大视差
    block_size: 匹配块大小
    P1, P2: 惩罚参数
    uniqueness_ratio: 唯一性比率
    speckle_window_size: 斑点滤波窗口大小
    speckle_range: 斑点滤波范围
    disp12_max_diff: 左右视差一致性检查的最大差异

输出:
    disparity_map: 视差图

1. 初始化:
    - 计算图像的宽度和高度: width, height
    - 创建视差图: disparity_map = zeros(height, width)
    - 创建代价矩阵: cost_matrix = zeros(height, width, max_disparity - min_disparity)
    - 创建聚合代价矩阵: aggregated_cost = zeros(height, width, max_disparity - min_disparity)

2. 计算初始代价:
    for each pixel (x, y) in left_image:
        for each disparity d in [min_disparity, max_disparity]:
            cost_matrix[y, x, d] = compute_matching_cost(left_image, right_image, x, y, d, block_size)

3. 代价聚合:
    for each direction in [left, right, up, down, ...]:
        for each pixel (x, y) in left_image:
            for each disparity d in [min_disparity, max_disparity]:
                aggregated_cost[y, x, d] += compute_aggregated_cost(cost_matrix, x, y, d, direction, P1, P2)

4. 视差计算:
    for each pixel (x, y) in left_image:
        disparity_map[y, x] = find_best_disparity(aggregated_cost, x, y, min_disparity, max_disparity)

5. 视差优化:
    - 唯一性检查: enforce_uniqueness(disparity_map, uniqueness_ratio)
    - 斑点滤波: remove_speckles(disparity_map, speckle_window_size, speckle_range)
    - 左右一致性检查: check_consistency(disparity_map, disp12_max_diff)

6. 返回视差图:
    return disparity_map
```

### 关键函数说明

- **compute_matching_cost**: 计算像素 `(x, y)` 在视差 `d` 下的匹配代价，通常使用SAD（Sum of Absolute Differences）或SSD（Sum of Squared Differences）。
- **compute_aggregated_cost**: 在给定方向上聚合代价，考虑相邻像素的视差连续性，使用动态规划方法。
- **find_best_disparity**: 在聚合代价矩阵中找到最佳视差。
- **enforce_uniqueness**: 确保视差的唯一性，避免模糊匹配。
- **remove_speckles**: 去除视差图中的小斑点噪声。
- **check_consistency**: 检查左右视差图的一致性，确保视差的准确性。

### 参数说明

- **P1, P2**: 控制视差平滑性的惩罚参数，P1 用于小视差变化，P2 用于大视差变化。
- **uniqueness_ratio**: 唯一性比率，用于过滤掉不唯一的匹配。
- **speckle_window_size**: 斑点滤波的窗口大小。
- **speckle_range**: 斑点滤波的范围。
- **disp12_max_diff**: 左右视差一致性检查的最大差异。

### 输出

- **disparity_map**: 计算得到的视差图，每个像素值表示该点的视差。

2. **匹配代价计算**：使用SAD（Sum of Absolute Differences）来计算左右图像中像素的匹配代价。
3. **视差计算**：通过最小化匹配代价来确定每个像素的视差。
4. **视差优化**：使用中值滤波来优化视差图。

以下是完整的`compute_disparity`函数的实现：

```rust
use image::{GrayImage, Luma};
use std::cmp;

fn compute_disparity(left: &GrayImage, right: &GrayImage) -> GrayImage {
    let (width, height) = left.dimensions();
    let max_disparity = 64; // 最大视差范围
    let mut disparity_map = GrayImage::new(width, height);

    for y in 0..height {
        for x in 0..width {
            let mut min_sad = u32::MAX;
            let mut best_disparity = 0;

            for d in 0..max_disparity {
                if x >= d {
                    let mut sad = 0;

                    // 计算SAD
                    for i in -1..=1 {
                        for j in -1..=1 {
                            let lx = x as i32 + i;
                            let ly = y as i32 + j;
                            let rx = (x as i32 - d) + i;
                            let ry = y as i32 + j;

                            if lx >= 0 && lx < width as i32 && ly >= 0 && ly < height as i32 &&
                               rx >= 0 && rx < width as i32 && ry >= 0 && ry < height as i32 {
                                let left_pixel = left.get_pixel(lx as u32, ly as u32)[0];
                                let right_pixel = right.get_pixel(rx as u32, ry as u32)[0];
                                sad += (left_pixel as i32 - right_pixel as i32).abs() as u32;
                            }
                        }
                    }

                    if sad < min_sad {
                        min_sad = sad;
                        best_disparity = d;
                    }
                }
            }

            disparity_map.put_pixel(x, y, Luma([best_disparity as u8]));
        }
    }

    // 视差优化：中值滤波
    let filtered_disparity_map = median_filter(&disparity_map, 3);
    filtered_disparity_map
}

fn median_filter(image: &GrayImage, window_size: u32) -> GrayImage {
    let (width, height) = image.dimensions();
    let mut filtered_image = GrayImage::new(width, height);
    let half_window = window_size as i32 / 2;

    for y in 0..height {
        for x in 0..width {
            let mut values = Vec::new();

            for i in -half_window..=half_window {
                for j in -half_window..=half_window {
                    let nx = x as i32 + i;
                    let ny = y as i32 + j;

                    if nx >= 0 && nx < width as i32 && ny >= 0 && ny < height as i32 {
                        values.push(image.get_pixel(nx as u32, ny as u32)[0]);
                    }
                }
            }

            values.sort();
            let median = values[values.len() / 2];
            filtered_image.put_pixel(x, y, Luma([median]));
        }
    }

    filtered_image
}

fn main() {
    // 示例：加载左右图像并计算视差图
    let left_image = image::open("left.png").unwrap().to_luma8();
    let right_image = image::open("right.png").unwrap().to_luma8();

    let disparity_map = compute_disparity(&left_image, &right_image);
    disparity_map.save("disparity.png").unwrap();
}
```

### 代码说明：
1. **匹配代价计算**：我们使用SAD（Sum of Absolute Differences）来计算左右图像中像素的匹配代价。SAD是通过计算左右图像中对应像素的绝对差值的和来衡量的。
2. **视差计算**：通过遍历每个像素，并在给定的最大视差范围内寻找最小SAD值，从而确定最佳视差。
3. **视差优化**：使用中值滤波来去除视差图中的噪声。中值滤波是一种非线性滤波方法，能够有效去除椒盐噪声。

### 注意事项：
- 代码中假设输入的左右图像已经经过校正，即左右图像在同一平面上。
- `max_disparity`是最大视差范围，可以根据实际情况调整。
- 中值滤波的窗口大小可以根据需要进行调整。

### 运行代码：
- 你需要准备左右两张图像（`left.png`和`right.png`），并确保它们在同一个目录下。
- 运行代码后，生成的视差图将保存为`disparity.png`。

这个实现是一个基础的视差计算流程，适用于简单的场景。对于更复杂的场景，可能需要更高级的算法和优化技术。

**SAD** 是 **Sum of Absolute Differences** 的缩写，中文意思是 **绝对差值和**。它是一种用于衡量两个图像块（或像素区域）之间相似度的简单方法。

### SAD 的计算方法：
对于两个图像块（或像素区域），SAD 的计算公式如下：

\[
\text{SAD} = \sum_{i,j} |I_{\text{left}}(i, j) - I_{\text{right}}(i, j)|
\]

其中：
- \( I_{\text{left}}(i, j) \) 是左图像中像素 \((i, j)\) 的灰度值。
- \( I_{\text{right}}(i, j) \) 是右图像中像素 \((i, j)\) 的灰度值。
- \( | \cdot | \) 表示绝对值。

### SAD 的意义：
- SAD 的值越小，表示两个图像块（或像素区域）越相似。
- SAD 的值越大，表示两个图像块（或像素区域）差异越大。

### 在视差计算中的作用：
在立体视觉中，SAD 用于计算左右图像中对应像素的匹配代价。通过比较左图像中的某个像素与右图像中一系列候选像素的 SAD 值，可以找到最匹配的像素，从而确定视差。

### 示例：
假设左图像中某个像素的灰度值为 100，右图像中某个候选像素的灰度值为 105，那么它们的绝对差值就是：

\[
|100 - 105| = 5
\]

如果比较的是一个 3x3 的图像块，那么 SAD 就是这 9 个像素的绝对差值的总和。

### 优点：
- 计算简单，易于实现。
- 对光照变化有一定的鲁棒性。

### 缺点：
- 对噪声敏感。
- 对于纹理较少的区域，匹配效果可能较差。

### 代码中的 SAD 计算：
在之前的代码中，SAD 是通过以下方式计算的：


1. **视差图计算**：使用OpenCV的`StereoSGBM`算法计算左右图像的视差图。
2. **点云生成**：根据视差图和相机内参，计算每个像素的三维坐标，并生成点云。
3. **点云显示**：使用`PyQt5`和`matplotlib`创建一个3D点云可视化窗口。
